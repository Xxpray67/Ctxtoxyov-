import math

def Mean(X, Y):
    """Вычисляет среднее арифметическое и среднее геометрическое."""
    AMean = (X + Y) / 2
    GMean = math.sqrt(X * Y)
    return AMean, GMean

# Задаем исходные значения A, B, C, D (можно заменить на input)
A, B, C, D = 4.0, 9.0, 16.0, 25.0

# Находим средние для пар (A, B), (A, C), (A, D)
pairs = [("A и B", A, B), ("A и C", A, C), ("A и D", A, D)]

for name, x, y in pairs:
    a_mean, g_mean = Mean(x, y)
    print(f"Пара {name} ({x}, {y}): AMean = {a_mean}, GMean = {g_mean}")


import math
def TrianglePS(a):
    P = 3 * a
    S = (a ** 2) * math.sqrt(3) / 4
    return P, S
sides = [4.0, 5.0, 1.0, 2.0]
for side in sides:
    perimeter, area = TrianglePS(side)
print (f"Сторона: {side} -> Периметр: {perimeter:.2f}, Площадь: {area:.2f}")

import math
def rect_inf(x1, y1, x2, y2):
    distance1 = abs(x2 - x1)
    distance2 = abs(y2 - y1)
    P = (distance1 + distance2) * 2
    S = distance1 * distance2
    return P, S
print (rect_inf(2, 3, 5, 4))
